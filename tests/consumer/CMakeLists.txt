cmake_minimum_required(VERSION 3.31)
project(polymarket_consumer LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -DPOLYMARKET_PARENT_FLAGS")
if(POLYMARKET_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(polymarket_cpp SOURCE_DIR "${POLYMARKET_SOURCE_DIR}")
    FetchContent_MakeAvailable(polymarket_cpp)
    if(POLYMARKET_BUILD_TESTS OR POLYMARKET_BUILD_EXAMPLES OR POLYMARKET_BUILD_APPS OR POLYMARKET_ENABLE_LTO)
        message(FATAL_ERROR "Embedded SDK defaults must be off")
    endif()
    get_directory_property(flags DIRECTORY "${POLYMARKET_SOURCE_DIR}" DEFINITION CMAKE_CXX_FLAGS_RELEASE)
    if(NOT flags STREQUAL "-O2 -DNDEBUG -DPOLYMARKET_PARENT_FLAGS")
        message(FATAL_ERROR "SDK overwrote parent compiler flags")
    endif()
else()
    find_package(polymarket 0.5.0 EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE polymarket::polymarket)
if(README_EXAMPLES_DIR)
    file(GLOB examples "${README_EXAMPLES_DIR}/*.cpp")
    foreach(example IN LISTS examples)
        get_filename_component(name "${example}" NAME_WE)
        add_executable("${name}" "${example}")
        target_link_libraries("${name}" PRIVATE polymarket::polymarket)
    endforeach()
endif()
